模拟交易系统架构设计：从信号扫描到自动执行

2026-08-26 Python自动化

实盘交易系统的核心难点不在策略，而在工程化。信号计算与自动执行之间的延迟、异常处理、状态一致性，每一项都可能让策略失效。本文分享一套完整的模拟交易系统架构，覆盖信号扫描、市场分析、模拟撮合与通知推送四大模块，代码可直接参考改造。

一、系统整体架构

整个系统采用生产者-消费者模型，模块间通过消息队列解耦，避免单点阻塞。

┌─────────────────────────────────────────────────────┐
│                    Scheduler (APScheduler)           │
│             每分钟触发一次 scan_job()                │
└─────────────────────────┬───────────────────────────┘
                          │
                          ▼
┌─────────────────────────────────────────────────────┐
│              SignalScanner (信号扫描模块)             │
│  读取行情 → 计算技术指标 → 生成交易信号 → 写入Redis   │
└─────────────────────────┬───────────────────────────┘
                          │ (Redis Stream: signals)
                          ▼
┌─────────────────────────────────────────────────────┐
│              MarketAnalyzer (市场分析模块)            │
│  接收信号 → 校验流动性/波动率 → 过滤无效信号 → 发布    │
└─────────────────────────┬───────────────────────────┘
                          │ (Redis Stream: validated)
                          ▼
┌─────────────────────────────────────────────────────┐
│            PaperTradingEngine (模拟交易引擎)          │
│  接收有效信号 → 模拟撮合 → 管理持仓/资金 → 更新状态    │
└─────────────────────────┬───────────────────────────┘
                          │ (状态变更事件)
                          ▼
┌─────────────────────────────────────────────────────┐
│            NotificationService (推送通知模块)         │
│  监听状态变更 → 格式化消息 → 推送至Server酱/Telegram  │
└─────────────────────────────────────────────────────┘

数据流向：行情源 → Redis → 各模块异步消费 → 状态库(SQLite/MySQL) → 通知渠道。

关键技术选型：

  • 调度：APScheduler（轻量，支持cron表达式）
  • 消息队列：Redis Stream（比List更适合消费者组，支持ack机制）
  • 状态存储：SQLite（模拟盘单机够用，实盘换PostgreSQL）
  • 行情源：ccxt（统一接口，支持120+交易所）
  • 通知：Server酱（微信推送）或Telegram Bot

二、模块职责与核心代码

2.1 信号扫描模块 (SignalScanner)

职责：定时拉取行情，计算指标，输出原始信号。只做计算，不做决策。

# scanner.py
import time
import ccxt
import pandas as pd
import redis
from apscheduler.schedulers.blocking import BlockingScheduler

class SignalScanner:
    def __init__(self, exchange_id='binance', symbol='BTC/USDT'):
        self.exchange = getattr(ccxt, exchange_id)({
            'enableRateLimit': True,
            'timeout': 10000,
        })
        self.symbol = symbol
        self.redis_client = redis.Redis(host='localhost', port=6379, db=0)
        self.stream_key = 'signals:raw'

    def fetch_ohlcv(self, timeframe='1m', limit=100):
        """获取K线数据，返回DataFrame"""
        data = self.exchange.fetch_ohlcv(self.symbol, timeframe=timeframe, limit=limit)
        df = pd.DataFrame(data, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        df['timestamp'] = pd.to_datetime(df['timestamp'], unit='ms')
        return df

    def calculate_signal(self, df):
        """基于均线交叉产生信号 - 仅示例，请替换为你的策略"""
        df['fast_ma'] = df['close'].rolling(window=5).mean()
        df['slow_ma'] = df['close'].rolling(window=20).mean()

        current_fast = df['fast_ma'].iloc[-1]
        current_slow = df['slow_ma'].iloc[-1]
        prev_fast = df['fast_ma'].iloc[-2]
        prev_slow = df['slow_ma'].iloc[-2]

        if prev_fast <= prev_slow and current_fast > current_slow:
            return 'BUY'
        elif prev_fast >= prev_slow and current_fast < current_slow:
            return 'SELL'
        return None

    def scan_once(self):
        """执行一次扫描并发布信号"""
        try:
            df = self.fetch_ohlcv()
            signal = self.calculate_signal(df)
            if signal:
                # 将信号写入Redis Stream
                event = {
                    'symbol': self.symbol,
                    'signal': signal,
                    'price': df['close'].iloc[-1],
                    'timestamp': int(time.time()),
                    'meta': {
                        'fast_ma': round(df['fast_ma'].iloc[-1], 2),
                        'slow_ma': round(df['slow_ma'].iloc[-1], 2),
                    }
                }
                self.redis_client.xadd(
                    self.stream_key,
                    event,
                    maxlen=1000  # 只保留最近1000条，防止内存膨胀
                )
                print(f"[SCANNER] Signal published: {event}")
            else:
                print(f"[SCANNER] No signal at {time.strftime('%H:%M:%S')}")
        except Exception as e:
            print(f"[SCANNER] Error: {e}")

    def run_scheduler(self, interval_minutes=1):
        """启动定时任务"""
        scheduler = BlockingScheduler()
        scheduler.add_job(
            self.scan_once,
            'interval',
            minutes=interval_minutes,
            id='scan_job',
            misfire_grace_time=30  # 任务错过30秒内仍执行
        )
        scheduler.start()

if __name__ == '__main__':
    scanner = SignalScanner()
    scanner.run_scheduler(interval_minutes=1)

2.2 市场分析模块 (MarketAnalyzer)

职责：对原始信号二次过滤。检查市场深度、波动率、交易量是否满足最低要求。防止在流动性枯竭时发出无效信号。

# analyzer.py
import redis
import time
import ccxt

class MarketAnalyzer:
    def __init__(self, exchange_id='binance', symbol='BTC/USDT'):
        self.exchange = getattr(ccxt, exchange_id)({
            'enableRateLimit': True,
        })
        self.symbol = symbol
        self.redis_client = redis.Redis(host='localhost', port=6379, db=0)
        self.input_stream = 'signals:raw'
        self.output_stream = 'signals:validated'
        self.group_name = 'market_analyzer'
        self.consumer_name = f'analyzer_{time.time()}'

    def check_liquidity(self, min_depth_usd=50000):
        """检查订单簿深度是否足够"""
        orderbook = self.exchange.fetch_order_book(self.symbol, limit=10)
        bid_depth = sum([item[0]*item[1] for item in orderbook['bids']])
        ask_depth = sum([item[0]*item[1] for item in orderbook['asks']])
        return min(bid_depth, ask_depth) > min_depth_usd

    def check_volatility(self, df, max_spread_pct=0.01):
        """检查买卖价差是否过大"""
        ticker = self.exchange.fetch_ticker(self.symbol)
        spread = (ticker['ask'] - ticker['bid']) / ticker['bid']
        return spread < max_spread_pct

    def process_signal(self, signal_data):
        """对信号进行过滤，返回是否通过"""
        symbol = signal_data['symbol']
        # 模拟获取实时ticker判断价差
        ticker = self.exchange.fetch_ticker(symbol)
        spread_pct = (ticker['ask'] - ticker['bid']) / ticker['bid']

        # 流动性检查（简化示例，实际应调用orderbook）
        if spread_pct > 0.005:  # 价差超过0.5%拒绝
            print(f"[ANALYZER] Rejected: spread too wide {spread_pct:.4f}")
            return False

        # 波动率检查（可用ATR或标准差，此处简化）
        if ticker['percentage'] and abs(ticker['percentage']) > 5:
            print(f"[ANALYZER] Rejected: too volatile {ticker['percentage']:.2f}%")
            return False

        return True

    def consume_and_publish(self):
        """消费原始信号，过滤后发布"""
        # 创建消费者组（如果不存在）
        try:
            self.redis_client.xgroup_create(
                self.input_stream, self.group_name, id='0', mkstream=True
            )
        except redis.exceptions.ResponseError:
            pass  # 组已存在

        while True:
            # 阻塞读取新消息
            messages = self.redis_client.xreadgroup(
                self.group_name, self.consumer_name,
                {self.input_stream: '>'}, count=5, block=5000
            )
            if not messages:
                continue

            for stream, entries in messages:
                for msg_id, data in entries:
                    # 注意：redis返回的data是bytes类型，需要解码
                    signal_data = {k.decode(): v.decode() if isinstance(v, bytes) else v
                                   for k, v in data.items()}
                    print(f"[ANALYZER] Processing: {signal_data}")

                    if self.process_signal(signal_data):
                        # 发布到验证后信号流
                        self.redis_client.xadd(
                            self.output_stream,
                            {**signal_data, 'validated_at': int(time.time())},
                            maxlen=1000
                        )
                        print(f"[ANALYZER] Signal validated, published to stream")
                    else:
                        print(f"[ANALYZER] Signal rejected: {signal_data['symbol']}")

                    # 确认消息已处理
                    self.redis_client.xack(self.input_stream, self.group_name, msg_id)

if __name__ == '__main__':
    analyzer = MarketAnalyzer()
    analyzer.consume_and_publish()

2.3 模拟交易引擎 (PaperTradingEngine)

职责：核心模块。维护虚拟账户、持仓、订单状态。关键设计：使用状态机模式管理订单生命周期，避免状态混乱。

# paper_engine.py
import sqlite3
import json
import time
import redis
from enum import Enum

class OrderStatus(Enum):
    PENDING = 'PENDING'
    FILLED = 'FILLED'
    REJECTED = 'REJECTED'
    CANCELLED = 'CANCELLED'

class PaperTradingEngine:
    def __init__(self, db_path='paper_trading.db', initial_balance=100000):
        self.conn = sqlite3.connect(db_path, check_same_thread=False)
        self.redis_client = redis.Redis(host='localhost', port=6379, db=0)
        self.input_stream = 'signals:validated'
        self.group_name = 'paper_engine'
        self.consumer_name = f'engine_{time.time()}'
        self.initial_balance = initial_balance
        self._init_db()

    def _init_db(self):
        """初始化数据库表"""
        cursor = self.conn.cursor()
        # 账户表
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS accounts (
                id INTEGER PRIMARY KEY,
                balance REAL NOT NULL,
                equity REAL NOT NULL,
                updated_at INTEGER
            )
        ''')
        # 持仓表
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS positions (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                symbol TEXT NOT NULL,
                side TEXT NOT NULL,
                quantity REAL NOT NULL,
                entry_price REAL NOT NULL,
                status TEXT NOT NULL,
                opened_at INTEGER,
                closed_at INTEGER
            )
        ''')
        # 订单表
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS orders (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                symbol TEXT NOT NULL,
                side TEXT NOT NULL,
                order_type TEXT NOT NULL,
                quantity REAL NOT NULL,
                price REAL NOT NULL,
                status TEXT NOT NULL,
                created_at INTEGER,
                filled_at INTEGER
            )
        ''')
        # 初始化账户
        cursor.execute('SELECT COUNT(*) FROM accounts')
        if cursor.fetchone()[0] == 0:
            cursor.execute(
                'INSERT INTO accounts (balance, equity, updated_at) VALUES (?, ?, ?)',
                (self.initial_balance, self.initial_balance, int(time.time()))
            )
        self.conn.commit()

    def _get_balance(self):
        cursor = self.conn.cursor()
        cursor.execute('SELECT balance, equity FROM accounts WHERE id=1')
        return cursor.fetchone()

    def _update_balance(self, balance, equity):
        cursor = self.conn.cursor()
        cursor.execute(
            'UPDATE accounts SET balance=?, equity=?, updated_at=? WHERE id=1',
            (balance, equity, int(time.time()))
        )
        self.conn.commit()

    def execute_order(self, symbol, side, quantity, price):
        """模拟撮合：立即全部成交（模拟盘简化处理）"""
        # 模拟滑点：成交价 = 信号价 * (1 + 滑点)
        slippage = 0.0005  # 0.05%
        fill_price = price * (1 + slippage if side == 'BUY' else 1 - slippage)

        # 计算所需资金
        cost = fill_price * quantity
        balance, equity = self._get_balance()

        if side == 'BUY':
            if cost > balance:
                print(f"[ENGINE] Order rejected: insufficient balance {balance:.2f} < {cost:.2f}")
                return OrderStatus.REJECTED.value
            new_balance = balance - cost
        else:  # SELL
            # 检查持仓
            cursor = self.conn.cursor()
            cursor.execute(
                'SELECT quantity FROM positions WHERE symbol=? AND side=? AND status=?',
                (symbol, 'BUY', 'OPEN')
            )
            pos = cursor.fetchone()
            if not pos or pos[0] < quantity:
                print(f"[ENGINE] Order rejected: insufficient position")
                return OrderStatus.REJECTED.value
            new_balance = balance + cost

        # 更新持仓（简化：只支持单向持仓）
        cursor = self.conn.cursor()
        if side == 'BUY':
            cursor.execute('''
                INSERT INTO positions (symbol, side, quantity, entry_price, status, opened_at)
                VALUES (?, 'BUY', ?, ?, 'OPEN', ?)
            ''', (symbol, quantity, fill_price, int(time.time())))
        else:
            # 平仓逻辑：计算盈亏
            cursor.execute(
                'SELECT entry_price, quantity FROM positions WHERE symbol=? AND side=? AND status=?',
                (symbol, 'BUY', 'OPEN')
            )
            entry = cursor.fetchone()
            pnl = (fill_price - entry[0]) * quantity
            new_balance += pnl  # 盈亏计入余额
            cursor.execute(
                'UPDATE positions SET status=?, closed_at=? WHERE symbol=? AND side=? AND status=?',
                ('CLOSED', int(time.time()), symbol, 'BUY', 'OPEN')
            )

        # 记录订单
        cursor.execute('''
            INSERT INTO orders (symbol, side, order_type, quantity, price, status, created_at, filled_at)
            VALUES (?, ?, 'MARKET', ?, ?, 'FILLED', ?, ?)
        ''', (symbol, side, quantity, fill_price, int(time.time()), int(time.time())))

        self._update_balance(new_balance, new_balance)
        self.conn.commit()

        # 发布状态变更事件
        self._publish_state_change(symbol, side, quantity, fill_price)
        print(f"[ENGINE] Order filled: {side} {quantity} {symbol} @ {fill_price:.2f}")
        return OrderStatus.FILLED.value

    def _publish_state_change(self, symbol, side, quantity, price):
        """发布状态变更到通知流"""
        event = {
            'type': 'ORDER_FILLED',
            'symbol': symbol,
            'side': side,
            'quantity': quantity,
            'price': price,
            'timestamp': int(time.time()),
        }
        self.redis_client.xadd('events:order_filled', event, maxlen=100)

    def consume_signals(self):
        """消费验证后的信号"""
        try:
            self.redis_client.xgroup_create(
                self.input_stream, self.group_name, id='0', mkstream=True
            )
        except redis.exceptions.ResponseError:
            pass

        while True:
            messages = self.redis_client.xreadgroup(
                self.group_name, self.consumer_name,
                {self.input_stream: '>'}, count=10, block=5000
            )
            if not messages:
                continue

            for stream, entries in messages:
                for msg_id, data in entries:
                    signal = {k.decode(): v.decode() if isinstance(v, bytes) else v
                              for k, v in data.items()}
                    print(f"[ENGINE] Processing signal: {signal}")

                    # 固定交易数量（示例）
                    quantity = 0.01
                    self.execute_order(
                        symbol=signal['symbol'],
                        side=signal['signal'],
                        quantity=quantity,
                        price=float(signal['price'])
                    )

                    self.redis_client.xack(self.input_stream, self.group_name, msg_id)

if __name__ == '__main__':
    engine = PaperTradingEngine()
    engine.consume_signals()
