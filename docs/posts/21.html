模拟交易系统架构设计：从信号扫描到自动执行

2026-09-12 Python自动化

做量化策略开发时，最痛苦的事情不是写策略逻辑，而是没有一个稳定的模拟环境来验证信号质量和执行效果。直接上实盘风险太大，纯回测又无法反映实时数据的真实表现。这篇文章分享一套我实际在用的模拟交易系统架构，从信号扫描到自动执行完整跑通。

整体架构

系统分为四个核心模块，通过事件驱动的方式串联：

┌─────────────┐     ┌─────────────┐     ┌─────────────┐     ┌─────────────┐
│  信号扫描    │────▶│  市场分析    │────▶│  模拟交易    │────▶│  推送通知    │
│  Scanner    │     │  Analyzer   │     │  Engine     │     │  Notifier   │
└─────────────┘     └─────────────┘     └─────────────┘     └─────────────┘
       │                   │                   │                   │
       ▼                   ▼                   ▼                   ▼
  行情数据源          指标计算库          虚拟账户            消息通道
  (REST/WS)          (pandas-ta)        (SQLite)          (Webhook)

数据流向很清晰：Scanner 负责定时拉取行情并生成原始信号，Analyzer 对信号做二次过滤和打分，Engine 根据分析结果执行模拟买卖并更新虚拟账户，Notifier 把关键事件推送到手机或群组。

模块之间通过一个轻量级的事件总线解耦，不直接互相调用。这样每个模块可以独立测试和替换。

事件总线设计

先定义事件总线和基础事件类型，这是整个系统的骨架：

import asyncio
import logging
from dataclasses import dataclass, field
from datetime import datetime
from typing import Callable, Any
from collections import defaultdict

logger = logging.getLogger(__name__)


@dataclass
class Event:
    topic: str
    payload: dict
    timestamp: datetime = field(default_factory=datetime.now)


class EventBus:
    """异步事件总线，模块间通过发布/订阅通信"""

    def __init__(self):
        self._subscribers: dict[str, list[Callable]] = defaultdict(list)

    def subscribe(self, topic: str, handler: Callable):
        self._subscribers[topic].append(handler)

    async def publish(self, topic: str, payload: dict):
        event = Event(topic=topic, payload=payload)
        handlers = self._subscribers.get(topic, [])
        if not handlers:
            return
        await asyncio.gather(
            *[self._safe_call(h, event) for h in handlers]
        )

    async def _safe_call(self, handler: Callable, event: Event):
        try:
            result = handler(event)
            if asyncio.iscoroutine(result):
                await result
        except Exception:
            logger.exception(f"Handler {handler.__name__} failed on {event.topic}")


bus = EventBus()

事件主题统一定义为常量，避免拼写错误：

class Topics:
    RAW_SIGNAL = "signal.raw"          # 扫描器产出原始信号
    ANALYZED_SIGNAL = "signal.analyzed" # 分析器产出过滤后的信号
    ORDER_FILLED = "order.filled"       # 模拟成交
    ORDER_REJECTED = "order.rejected"   # 订单被拒
    ACCOUNT_UPDATE = "account.update"   # 账户变动

信号扫描模块

Scanner 的职责很单一：按固定间隔拉取行情数据，跑一遍预设的信号规则，把命中的结果发布出去。

import aiohttp
import asyncio
from typing import Optional


class MarketDataFetcher:
    """行情数据获取，这里用 REST 轮询，实际可换成 WebSocket"""

    def __init__(self, base_url: str):
        self.base_url = base_url
        self._session: Optional[aiohttp.ClientSession] = None

    async def start(self):
        self._session = aiohttp.ClientSession()

    async def stop(self):
        if self._session:
            await self._session.close()

    async def fetch_klines(self, symbol: str, interval: str = "1m", limit: int = 100) -> list:
        url = f"{self.base_url}/api/v1/klines"
        params = {"symbol": symbol, "interval": interval, "limit": limit}
        async with self._session.get(url, params=params) as resp:
            resp.raise_for_status()
            return await resp.json()


class SignalScanner:
    """信号扫描器：定时拉行情，跑规则，发信号"""

    def __init__(self, fetcher: MarketDataFetcher, bus: EventBus,
                 symbols: list[str], interval_sec: int = 60):
        self.fetcher = fetcher
        self.bus = bus
        self.symbols = symbols
        self.interval_sec = interval_sec
        self._rules: list[Callable] = []

    def register_rule(self, rule_fn: Callable):
        """注册信号规则，规则函数接收 klines 返回 signal dict 或 None"""
        self._rules.append(rule_fn)

    async def run(self):
        while True:
            try:
                await self._scan_once()
            except Exception:
                logger.exception("Scan cycle failed")
            await asyncio.sleep(self.interval_sec)

    async def _scan_once(self):
        tasks = [self._scan_symbol(s) for s in self.symbols]
        await asyncio.gather(*tasks)

    async def _scan_symbol(self, symbol: str):
        klines = await self.fetcher.fetch_klines(symbol)
        if not klines or len(klines) < 20:
            return
        for rule in self._rules:
            signal = rule(symbol, klines)
            if signal:
                await self.bus.publish(Topics.RAW_SIGNAL, signal)
                logger.info(f"Raw signal: {signal}")

一个简单的规则示例，检测放量突破：

def volume_breakout_rule(symbol: str, klines: list) -> dict | None:
    """最近一根K线成交量超过前20根均量的2倍，且收盘价创近期新高"""
    closes = [float(k[4]) for k in klines]
    volumes = [float(k[5]) for k in klines]

    avg_vol = sum(volumes[-21:-1]) / 20
    last_vol = volumes[-1]
    last_close = closes[-1]
    recent_high = max(closes[-20:-1])

    if last_vol > avg_vol * 2 and last_close > recent_high:
        return {
            "symbol": symbol,
            "type": "volume_breakout",
            "price": last_close,
            "volume_ratio": round(last_vol / avg_vol, 2),
            "ts": klines[-1][0],
        }
    return None

市场分析模块

Analyzer 订阅原始信号，做进一步过滤。比如检查趋势方向、计算波动率、给出信号评分。只有评分超过阈值的信号才会被放行到交易引擎。

import numpy as np


class MarketAnalyzer:
    """对原始信号做二次过滤和评分"""

    def __init__(self, bus: EventBus, min_score: float = 0.6):
        self.bus = bus
        self.min_score = min_score
        bus.subscribe(Topics.RAW_SIGNAL, self.on_raw_signal)

    async def on_raw_signal(self, event: Event):
        signal = event.payload
        score = self._score(signal)
        if score < self.min_score:
            logger.debug(f"Signal {signal['symbol']} scored {score:.2f}, filtered")
            return
        signal["score"] = round(score, 3)
        await self.bus.publish(Topics.ANALYZED_SIGNAL, signal)

    def _score(self, signal: dict) -> float:
        """多因子打分，返回 0~1"""
        scores = []

        # 因子1：量比越高越好，3倍以上满分
        vol_ratio = signal.get("volume_ratio", 1.0)
        scores.append(min(vol_ratio / 3.0, 1.0))

        # 因子2：信号类型权重
        type_weight = {"volume_breakout": 0.8, "ma_cross": 0.6}
        scores.append(type_weight.get(signal["type"], 0.5))

        # 因子3：价格动量（可扩展）
        # 这里留空，实际接入更多数据后补充

        return float(np.mean(scores))

这个模块的设计要点是：打分逻辑和信号生成逻辑分离。规则负责"发现"，分析器负责"判断"。这样调整过滤条件时不需要动扫描器。

模拟交易引擎

引擎是核心模块，维护虚拟账户、持仓、订单，处理成交逻辑。数据落地用 SQLite，轻量且方便查询。

import sqlite3
from dataclasses import dataclass


@dataclass
class Position:
    symbol: str
    quantity: float
    avg_price: float

    @property
    def cost(self) -> float:
        return self.quantity * self.avg_price


class VirtualAccount:
    """虚拟账户，记录余额、持仓、成交历史"""

    def __init__(self, db_path: str = "paper_trading.db",
                 initial_balance: float = 100_000):
        self.conn = sqlite3.connect(db_path)
        self._init_tables()
        self._ensure_account(initial_balance)

    def _init_tables(self):
        cur = self.conn.cursor()
        cur.executescript("""
            CREATE TABLE IF NOT EXISTS account (
                id INTEGER PRIMARY KEY CHECK (id = 1),
                balance REAL NOT NULL,
                frozen REAL DEFAULT 0
            );
            CREATE TABLE IF NOT EXISTS positions (
                symbol TEXT PRIMARY KEY,
                quantity REAL NOT NULL,
                avg_price REAL NOT NULL
            );
            CREATE TABLE IF NOT EXISTS trades (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                symbol TEXT NOT NULL,
                side TEXT NOT NULL,
                price REAL NOT NULL,
                quantity REAL NOT NULL,
                fee REAL NOT NULL,
                ts DATETIME DEFAULT CURRENT_TIMESTAMP
            );
        """)
        self.conn.commit()

    def _ensure_account(self, initial_balance: float):
        cur = self.conn.cursor()
        cur.execute("SELECT COUNT(*) FROM account")
        if cur.fetchone()[0] == 0:
            cur.execute("INSERT INTO account (id, balance) VALUES (1, ?)",
                        (initial_balance,))
            self.conn.commit()

    @property
    def balance(self) -> float:
        cur = self.conn.cursor()
        cur.execute("SELECT balance FROM account WHERE id = 1")
        return cur.fetchone()[0]

    def get_position(self, symbol: str) -> Position | None:
        cur = self.conn.cursor()
        cur.execute("SELECT quantity, avg_price FROM positions WHERE symbol = ?",
                    (symbol,))
        row = cur.fetchone()
        if row and row[0] > 0:
            return Position(symbol, row[0], row[1])
        return None

    def execute_buy(self, symbol: str, price: float,
                    quantity: float, fee_rate: float = 0.001) -> bool:
        cost = price * quantity
        fee = cost * fee_rate
        total = cost + fee
        if total > self.balance:
            return False

        cur = self.conn.cursor()
        cur.execute("UPDATE account SET balance = balance - ? WHERE id = 1",
                    (total,))

        pos = self.get_position(symbol)
        if pos:
            new_qty = pos.quantity + quantity
            new_avg = (pos.cost + cost) / new_qty
            cur.execute("UPDATE positions SET quantity = ?, avg_price = ? WHERE symbol = ?",
                        (new_qty, new_avg, symbol))
        else:
            cur.execute("INSERT INTO positions (symbol, quantity, avg_price) VALUES (?, ?, ?)",
                        (symbol, quantity, price))

        cur.execute("INSERT INTO trades (symbol, side, price, quantity, fee) VALUES (?, 'BUY', ?, ?, ?)",
                    (symbol, price, quantity, fee))
        self.conn.commit()
        return True

    def execute_sell(self, symbol: str, price: float,
                     quantity: float, fee_rate: float = 0.001) -> bool:
        pos = self.get_position(symbol)
        if not pos or pos.quantity < quantity:
            return False

        proceeds = price * quantity
        fee = proceeds * fee_rate
        net = proceeds - fee

        cur = self.conn.cursor()
        cur.execute("UPDATE account SET balance = balance + ? WHERE id = 1", (net,))

        remaining = pos.quantity - quantity
        if remaining <= 0:
            cur.execute("DELETE FROM positions WHERE symbol = ?", (symbol,))
        else:
            cur.execute("UPDATE positions SET quantity = ? WHERE symbol = ?",
                        (remaining, symbol))

        cur.execute("INSERT INTO trades (symbol, side, price, quantity, fee) VALUES (?, 'SELL', ?, ?, ?)",
                    (symbol, price, quantity, fee))
        self.conn.commit()
        return True

交易引擎订阅分析后的信号，根据信号类型决定买卖方向和仓位：

class TradingEngine:
    """模拟交易引擎：接收分析信号，执行模拟买卖"""

    def __init__(self, bus: EventBus, account: VirtualAccount,
                 position_size_pct: float = 0.1):
        self.bus = bus
        self.account = account
        self.position_size_pct = position_size_pct
        bus.subscribe(Topics.ANALYZED_SIGNAL, self.on_signal)

    async def on_signal(self, event: Event):
        signal = event.payload
        symbol = signal["symbol"]
        price = signal["price"]
        score = signal.get("score", 0.5)

        # 仓位大小随评分线性调整
        alloc = self.account.balance * self.position_size_pct * score
        quantity = round(alloc / price, 6)

        if quantity <= 0:
            return

        success = self.account.execute_buy(symbol, price, quantity)
        if success:
            await self.bus.publish(Topics.ORDER_FILLED, {
                "symbol": symbol, "side": "BUY",
                "price": price, "quantity": quantity,
                "score": score,
            })
        else:
            await self.bus.publish(Topics.ORDER_REJECTED, {
                "symbol": symbol, "reason": "insufficient_balance",
            })

推送通知模块

通知模块订阅成交和拒单事件，通过 Webhook 推送到手机或群组。这里用飞书/钉钉的 Webhook 格式举例：

class Notifier:
    """事件推送到 Webhook"""

    def __init__(self, bus: EventBus, webhook_url: str):
        self.webhook_url = webhook_url
        self._session: Optional[aiohttp.ClientSession] = None
        bus.subscribe(Topics.ORDER_FILLED, self.on_filled)
        bus.subscribe(Topics.ORDER_REJECTED, self.on_rejected)

    async def start(self):
        self._session = aiohttp.ClientSession()

    async def stop(self):
        if self._session:
            await self._session.close()

    async def on_filled(self, event: Event):
        d = event.payload
        text = (f"[模拟成交] {d['side']} {d['symbol']}\n"
                f"价格: {d['price']}  数量: {d['quantity']}\n"
                f"信号评分: {d.get('score', 'N/A')}")
        await self._send(text)

    async def on_rejected(self, event: Event):
        d = event.payload
        text = f"[订单拒绝] {d['symbol']}  原因: {d['reason']}"
        await self._send(text)

    async def _send(self, text: str):
        if not self._session:
            return
        payload = {"msg_type": "text", "content": {"text": text}}
        try:
            async with self._session.post(self.webhook_url, json=payload) as resp:
                if resp.status != 200:
                    logger.warning(f"Webhook returned {resp.status}")
        except Exception:
            logger.exception("Failed to send notification")

主入口与启动

把所有模块组装起来：

async def main():
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
    )

    fetcher = MarketDataFetcher(base_url="https://api.example.com")
    await fetcher.start()

    scanner = SignalScanner(
        fetcher=fetcher,
        bus=bus,
        symbols=["BTCUSDT", "ETHUSDT"],
        interval_sec=60,
    )
    scanner.register_rule(volume_breakout_rule)

    analyzer = MarketAnalyzer(bus=bus, min_score=0.6)
    account = VirtualAccount(db_path="paper.db", initial_balance=100_000)
    engine = TradingEngine(bus=bus, account=account, position_size_pct=0.1)

    notifier = Notifier(bus=bus, webhook_url="https://hook.example.com/xxx")
    await notifier.start()

    try:
        await scanner.run()
    finally:
        await fetcher.stop()
        await notifier.stop()


if __name__ == "__main__":
    asyncio.run(main())
