模拟交易系统架构设计：从信号扫描到自动执行

2026-09-20 Python自动化

做量化策略开发时，最头疼的问题不是策略本身，而是验证环节。回测跑得再好，实盘前总得有个过渡阶段——模拟交易。这篇文章拆解一套我实际在用的模拟交易系统架构，四个核心模块：信号扫描、市场分析、模拟交易引擎、推送通知。整套系统用 Python 实现，可以跑在单机上，也可以拆成微服务。

模块职责划分

信号扫描模块

核心任务只有一个：在合适的时间点，判断是否满足开仓条件。不要在这个模块里做仓位计算、不要做风控，保持单一职责。

import pandas as pd
import numpy as np
from dataclasses import dataclass
from datetime import datetime
from typing import Optional
import logging

logger = logging.getLogger(__name__)

@dataclass
class Signal:
    symbol: str
    direction: str  # 'long' or 'short'
    price: float
    timestamp: datetime
    strategy: str
    metadata: dict

class SignalScanner:
    def __init__(self, data_fetcher, strategies: list):
        self.data_fetcher = data_fetcher
        self.strategies = strategies

    def scan(self, symbols: list) -> list[Signal]:
        signals = []
        for symbol in symbols:
            try:
                df = self.data_fetcher.get_kline(symbol, interval='1h', limit=200)
                if df is None or len(df) < 50:
                    continue
                for strategy in self.strategies:
                    sig = strategy.check(symbol, df)
                    if sig:
                        signals.append(sig)
                        logger.info(f"信号触发: {symbol} {sig.direction} @ {sig.price}")
            except Exception as e:
                logger.error(f"扫描 {symbol} 失败: {e}")
        return signals

策略基类定义：

class BaseStrategy:
    name = "base"

    def check(self, symbol: str, df: pd.DataFrame) -> Optional[Signal]:
        raise NotImplementedError

class MaCrossStrategy(BaseStrategy):
    name = "ma_cross"

    def __init__(self, fast=10, slow=30):
        self.fast = fast
        self.slow = slow

    def check(self, symbol, df):
        df = df.copy()
        df['ma_fast'] = df['close'].rolling(self.fast).mean()
        df['ma_slow'] = df['close'].rolling(self.slow).mean()
        last, prev = df.iloc[-1], df.iloc[-2]

        # 金叉
        if prev['ma_fast'] <= prev['ma_slow'] and last['ma_fast'] > last['ma_slow']:
            return Signal(
                symbol=symbol, direction='long',
                price=last['close'], timestamp=datetime.now(),
                strategy=self.name,
                metadata={'ma_fast': last['ma_fast'], 'ma_slow': last['ma_slow']}
            )
        # 死叉
        if prev['ma_fast'] >= prev['ma_slow'] and last['ma_fast'] < last['ma_slow']:
            return Signal(
                symbol=symbol, direction='short',
                price=last['close'], timestamp=datetime.now(),
                strategy=self.name,
                metadata={'ma_fast': last['ma_fast'], 'ma_slow': last['ma_slow']}
            )
        return None

这里的关键设计：策略只负责"发现信号"，不负责"决定是否交易"。后续模块会做二次过滤。

市场分析模块

信号出来了不代表能直接下单。需要分析模块提供市场环境判断——当前波动率是否适合开仓、趋势强度如何、是否处于异常状态。

class MarketAnalyzer:
    def __init__(self, data_fetcher):
        self.data_fetcher = data_fetcher

    def analyze(self, symbol: str) -> dict:
        df = self.data_fetcher.get_kline(symbol, interval='1h', limit=100)
        if df is None or len(df) < 50:
            return {'valid': False}

        close = df['close']
        high, low = df['high'], df['low']

        # ATR 波动率
        tr = pd.concat([
            high - low,
            (high - close.shift()).abs(),
            (low - close.shift()).abs()
        ], axis=1).max(axis=1)
        atr = tr.rolling(14).mean().iloc[-1]
        atr_pct = atr / close.iloc[-1]

        # ADX 趋势强度
        adx = self._calc_adx(df, 14)

        # 布林带宽度
        ma20 = close.rolling(20).mean()
        std20 = close.rolling(20).std()
        bb_width = (4 * std20 / ma20).iloc[-1]

        return {
            'valid': True,
            'atr_pct': round(atr_pct, 4),
            'adx': round(adx, 2),
            'bb_width': round(bb_width, 4),
            'trend_strength': 'strong' if adx > 25 else 'weak',
            'volatility': 'high' if atr_pct > 0.02 else 'normal'
        }

    def _calc_adx(self, df, period=14):
        high, low, close = df['high'], df['low'], df['close']
        plus_dm = high.diff()
        minus_dm = -low.diff()
        plus_dm[plus_dm < 0] = 0
        minus_dm[minus_dm < 0] = 0

        tr = pd.concat([
            high - low,
            (high - close.shift()).abs(),
            (low - close.shift()).abs()
        ], axis=1).max(axis=1)

        atr = tr.rolling(period).mean()
        plus_di = 100 * (plus_dm.rolling(period).mean() / atr)
        minus_di = 100 * (minus_dm.rolling(period).mean() / atr)
        dx = 100 * (plus_di - minus_di).abs() / (plus_di + minus_di)
        return dx.rolling(period).mean().iloc[-1]

分析模块的输出是一个"市场状态字典"，交易引擎根据这个字典决定是否放行信号。

模拟交易引擎

这是整个系统最核心的部分。要处理订单生命周期、持仓管理、盈亏计算、风控。

from dataclasses import dataclass, field
from enum import Enum

class OrderStatus(Enum):
    PENDING = 'pending'
    FILLED = 'filled'
    REJECTED = 'rejected'
    CANCELLED = 'cancelled'

@dataclass
class Position:
    symbol: str
    direction: str
    entry_price: float
    size: float
    entry_time: datetime
    stop_loss: float = 0.0
    take_profit: float = 0.0

    def unrealized_pnl(self, current_price: float) -> float:
        if self.direction == 'long':
            return (current_price - self.entry_price) * self.size
        return (self.entry_price - current_price) * self.size

@dataclass
class Order:
    symbol: str
    direction: str
    price: float
    size: float
    status: OrderStatus = OrderStatus.PENDING
    created_at: datetime = field(default_factory=datetime.now)
    filled_at: Optional[datetime] = None

class PaperTradingEngine:
    def __init__(self, initial_balance=100000, commission_rate=0.0005,
                 max_positions=5, risk_per_trade=0.02):
        self.balance = initial_balance
        self.initial_balance = initial_balance
        self.commission_rate = commission_rate
        self.max_positions = max_positions
        self.risk_per_trade = risk_per_trade
        self.positions: dict[str, Position] = {}
        self.orders: list[Order] = []
        self.closed_trades: list[dict] = []

    def execute_signal(self, signal: Signal, market_state: dict) -> Optional[Order]:
        # 风控检查
        if not self._risk_check(signal, market_state):
            return None

        size = self._calc_position_size(signal)
        if size <= 0:
            return None

        order = Order(
            symbol=signal.symbol, direction=signal.direction,
            price=signal.price, size=size
        )

        # 模拟即时成交（可改为滑点模型）
        order.status = OrderStatus.FILLED
        order.filled_at = datetime.now()
        self.orders.append(order)

        # 更新持仓
        self._update_position(order)
        return order

    def _risk_check(self, signal, market_state) -> bool:
        # 持仓数量限制
        if len(self.positions) >= self.max_positions:
            return False
        # 同标的同方向不重复开仓
        if signal.symbol in self.positions:
            return False
        # 波动率过高时拒绝开仓
        if market_state.get('atr_pct', 0) > 0.05:
            return False
        return True

    def _calc_position_size(self, signal) -> float:
        risk_amount = self.balance * self.risk_per_trade
        # 简化处理：按固定比例
        size = risk_amount / signal.price
        return round(size, 4)

    def _update_position(self, order: Order):
        pos = Position(
            symbol=order.symbol, direction=order.direction,
            entry_price=order.price, size=order.size,
            entry_time=order.filled_at
        )
        self.positions[order.symbol] = pos
        cost = order.price * order.size * self.commission_rate
        self.balance -= cost

    def close_position(self, symbol: str, price: float) -> Optional[dict]:
        pos = self.positions.pop(symbol, None)
        if not pos:
            return None
        pnl = pos.unrealized_pnl(price)
        commission = price * pos.size * self.commission_rate
        self.balance += pnl - commission
        trade = {
            'symbol': symbol, 'direction': pos.direction,
            'entry': pos.entry_price, 'exit': price,
            'size': pos.size, 'pnl': round(pnl - commission, 2),
            'entry_time': pos.entry_time, 'exit_time': datetime.now()
        }
        self.closed_trades.append(trade)
        return trade

    def get_stats(self) -> dict:
        total_pnl = sum(t['pnl'] for t in self.closed_trades)
        wins = [t for t in self.closed_trades if t['pnl'] > 0]
        win_rate = len(wins) / len(self.closed_trades) if self.closed_trades else 0
        return {
            'balance': round(self.balance, 2),
            'total_pnl': round(total_pnl, 2),
            'return_pct': round((self.balance - self.initial_balance) / self.initial_balance * 100, 2),
            'total_trades': len(self.closed_trades),
            'win_rate': round(win_rate * 100, 2)
        }

引擎的设计要点：

  1. 风控前置：execute_signal 入口就做检查，不满足直接返回 None
  2. 仓位计算独立：_calc_position_size 可以换成凯利公式、固定比例、波动率倒数等
  3. 成交模型可插拔：目前是即时成交，可以扩展成滑点模型、部分成交模型

推送通知模块

信号和成交结果需要及时推送到手机或群组。推荐用 Server酱、钉钉 Webhook 或 Telegram Bot。

import requests
import json

class Notifier:
    def __init__(self, channels: list):
        self.channels = channels

    def send(self, title: str, content: str):
        for ch in self.channels:
            try:
                ch.send(title, content)
            except Exception as e:
                logging.error(f"推送失败 {ch.__class__.__name__}: {e}")

class DingTalkChannel:
    def __init__(self, webhook: str):
        self.webhook = webhook

    def send(self, title, content):
        payload = {
            "msgtype": "markdown",
            "markdown": {
                "title": title,
                "text": f"### {title}\n\n{content}"
            }
        }
        requests.post(self.webhook, json=payload, timeout=5)

class TelegramChannel:
    def __init__(self, bot_token: str, chat_id: str):
        self.url = f"https://api.telegram.org/bot{bot_token}/sendMessage"
        self.chat_id = chat_id

    def send(self, title, content):
        requests.post(self.url, json={
            "chat_id": self.chat_id,
            "text": f"*{title}*\n\n{content}",
            "parse_mode": "Markdown"
        }, timeout=5)


def format_signal(sig: Signal, market: dict) -> str:
    return (
        f"**标的**: {sig.symbol}\n"
        f"**方向**: {sig.direction}\n"
        f"**价格**: {sig.price}\n"
        f"**策略**: {sig.strategy}\n"
        f"**波动率**: {market.get('atr_pct', 'N/A')}\n"
        f"**趋势强度**: {market.get('trend_strength', 'N/A')}"
    )


def format_trade(trade: dict) -> str:
    emoji = "🟢" if trade['pnl'] > 0 else "🔴"
    return (
        f"{emoji} **平仓** {trade['symbol']}\n"
        f"方向: {trade['direction']}\n"
        f"入场: {trade['entry']} → 出场: {trade['exit']}\n"
        f"盈亏: {trade['pnl']}"
    )

class TradingSystem: def init(self, scanner, analyzer, engine, notifier): self.scanner = scanner self.analyzer = analyzer self.engine = engine self.notifier = notifier self.watch_symbols = []

def run_scan_cycle(self):
    signals = self.scanner.scan(self.watch_symbols)
    for sig in signals:
        market = self.analyzer.analyze(sig.symbol)
        if not market.get('valid'):
            continue

        order = self.engine.execute_signal(sig, market)
        if order:
            self.notifier.send(
                f"模拟开仓 - {sig.symbol}",
                format_signal(sig, market)
            )

def check_positions(self):
    """检查持仓，触发止盈止损"""
    for symbol in list(self.engine.positions.keys()):
        pos = self.engine.positions[symbol]
        df = self.scanner.data_fetcher.get_kline(symbol, '1h', 5)
        if df is None:
            continue
        current = df['close'].iloc[-1]

        # 简单止损逻辑：亏损超过 3% 平仓
        pnl_pct = pos.unrealized_pnl(current) / (pos.entry_price * pos.size)
        if pnl_pct < -0.03:
            trade = self.engine.close_position(symbol, current)
            if trade:
                self.notifier.send("模拟平仓", format_trade(trade))
