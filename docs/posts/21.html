模拟交易系统架构设计：从信号扫描到自动执行

2026-08-29 Python自动化

实盘交易系统与模拟交易系统最大的区别在于：前者需要处理券商接口、风控、滑点、成交回报等复杂逻辑，而后者可以专注于策略逻辑本身。但模拟交易系统并非简单地把信号打印出来——它需要一套完整的架构来支撑信号产生、分析、执行和通知的闭环。

这篇文章分享一个我在生产环境中使用的模拟交易系统架构，包含四个核心模块：信号扫描、市场分析、模拟交易引擎、推送通知。整个系统用 Python 实现，模块间通过消息队列解耦，支持横向扩展。

模块一：信号扫描模块

信号扫描是整个系统的入口。它负责监听行情数据，计算技术指标，生成原始信号。我推荐使用 APScheduler 做定时任务，配合 Redis Pub/Sub 或 Kafka 做事件驱动。

# scanner.py
import asyncio
from dataclasses import dataclass, field
from typing import Dict, List
import pandas as pd
import numpy as np
from apscheduler.schedulers.asyncio import AsyncIOScheduler
import redis.asyncio as redis

@dataclass
class Signal:
    symbol: str
    direction: str  # 'long' or 'short'
    strength: float  # 0.0 ~ 1.0
    timestamp: float
    metadata: Dict = field(default_factory=dict)

class SignalScanner:
    def __init__(self, redis_url: str = "redis://localhost:6379"):
        self.redis_client = redis.from_url(redis_url)
        self.scheduler = AsyncIOScheduler()
        self.strategies = {}  # 注册的策略字典

    def register_strategy(self, name: str, strategy):
        """注册策略到扫描器"""
        self.strategies[name] = strategy

    async def scan(self, symbol: str):
        """扫描单个标的，生成原始信号"""
        # 获取K线数据（假设从数据库或API获取）
        df = await self._get_klines(symbol)

        signals = []
        for name, strategy in self.strategies.items():
            sig = strategy.generate(df, symbol)
            if sig is not None:
                signals.append(sig)
                # 原始信号发送到Redis队列，供Analyzer消费
                await self.redis_client.publish(
                    'raw_signals',
                    json.dumps({
                        'strategy': name,
                        'signal': sig.__dict__
                    })
                )
        return signals

    async def _get_klines(self, symbol: str) -> pd.DataFrame:
        """从数据源获取K线（示例：从MySQL读取）"""
        import aiomysql
        conn = await aiomysql.connect(
            host='localhost', user='quant', password='pass', db='market'
        )
        async with conn.cursor() as cur:
            await cur.execute(
                "SELECT ts, open, high, low, close, volume FROM klines "
                "WHERE symbol=%s ORDER BY ts DESC LIMIT 500",
                (symbol,)
            )
            rows = await cur.fetchall()
        conn.close()
        df = pd.DataFrame(rows, columns=['ts', 'open', 'high', 'low', 'close', 'volume'])
        return df.sort_values('ts')

    def start(self):
        """启动定时扫描任务"""
        symbols = ['BTCUSDT', 'ETHUSDT', 'AAPL']  # 实际从配置读取

        for symbol in symbols:
            self.scheduler.add_job(
                self.scan, 'interval', minutes=5,
                args=[symbol], id=f'scan_{symbol}'
            )
        self.scheduler.start()

策略接口设计：

# strategy_base.py
from abc import ABC, abstractmethod
import pandas as pd

class BaseStrategy(ABC):
    @abstractmethod
    def generate(self, df: pd.DataFrame, symbol: str):
        """输入K线数据，输出Signal或None"""
        pass

# 示例策略：双均线交叉
class DualMAStrategy(BaseStrategy):
    def __init__(self, fast: int = 5, slow: int = 20):
        self.fast = fast
        self.slow = slow

    def generate(self, df: pd.DataFrame, symbol: str):
        if len(df) < self.slow + 5:
            return None

        df = df.copy()
        df['ma_fast'] = df['close'].rolling(self.fast).mean()
        df['ma_slow'] = df['close'].rolling(self.slow).mean()

        # 金叉
        if df['ma_fast'].iloc[-2] <= df['ma_slow'].iloc[-2] and \
           df['ma_fast'].iloc[-1] > df['ma_slow'].iloc[-1]:
            return Signal(
                symbol=symbol,
                direction='long',
                strength=0.7,
                timestamp=df['ts'].iloc[-1],
                metadata={'strategy': 'dual_ma'}
            )
        return None

模块二：市场分析模块

信号扫描会产生大量噪音信号。分析模块的作用是过滤掉这些噪音，结合市场大环境判断信号是否有效。

# analyzer.py
import json
import asyncio
import redis.asyncio as redis
from typing import Dict, Any

class MarketAnalyzer:
    def __init__(self, redis_url: str = "redis://localhost:6379"):
        self.redis = redis.from_url(redis_url)
        self.pubsub = self.redis.pubsub()
        self.valid_signals = []

    async def _get_market_regime(self, symbol: str) -> str:
        """判断市场状态：trending / ranging / volatile"""
        # 用ATR和ADX指标判断
        df = await self._get_klines(symbol)
        atr = self._calc_atr(df, period=14)
        adx = self._calc_adx(df, period=14)

        if adx > 25:
            return 'trending'
        elif atr > df['close'].mean() * 0.03:
            return 'volatile'
        else:
            return 'ranging'

    async def analyze(self, raw_signal: Dict[str, Any]) -> bool:
        """分析信号是否有效"""
        symbol = raw_signal['signal']['symbol']
        direction = raw_signal['signal']['direction']

        # 1. 市场状态过滤：震荡市不追多/空
        regime = await self._get_market_regime(symbol)
        if regime == 'ranging' and direction == 'trend_following':
            return False

        # 2. 多周期验证：检查4h和1d趋势是否一致
        trend_4h = await self._get_trend(symbol, timeframe='4h')
        trend_1d = await self._get_trend(symbol, timeframe='1d')
        if trend_4h != trend_1d:
            return False

        # 3. 成交量确认
        volume_ratio = await self._get_volume_ratio(symbol)
        if volume_ratio < 1.5:
            return False

        # 通过所有检查，标记为有效信号
        raw_signal['validated'] = True
        raw_signal['analysis'] = {
            'regime': regime,
            'trend_4h': trend_4h,
            'trend_1d': trend_1d,
            'volume_ratio': volume_ratio
        }

        # 发布到交易引擎队列
        await self.redis.publish('validated_signals', json.dumps(raw_signal))
        return True

    async def run(self):
        """订阅原始信号队列并处理"""
        await self.pubsub.subscribe('raw_signals')
        async for message in self.pubsub.listen():
            if message['type'] != 'message':
                continue
            try:
                signal = json.loads(message['data'])
                await self.analyze(signal)
            except Exception as e:
                print(f"Analysis error: {e}")

模块三：模拟交易引擎

这是系统的核心。它模拟真实交易环境，维护虚拟账户、持仓、订单状态，并计算盈亏。

# paper_engine.py
import asyncio
from dataclasses import dataclass, field
from typing import Dict, List, Optional
import json
import time
import redis.asyncio as redis

@dataclass
class Position:
    symbol: str
    direction: str
    entry_price: float
    quantity: float
    entry_time: float
    unrealized_pnl: float = 0.0

@dataclass
class Order:
    id: str
    symbol: str
    direction: str
    quantity: float
    price: float
    status: str  # 'pending', 'filled', 'cancelled'
    create_time: float
    fill_time: Optional[float] = None

class PaperTradingEngine:
    def __init__(self, initial_capital: float = 100000.0):
        self.capital = initial_capital
        self.positions: Dict[str, Position] = {}
        self.orders: List[Order] = []
        self.redis = redis.from_url("redis://localhost:6379")
        self.pubsub = self.redis.pubsub()
        self.risk_limit = 0.02  # 单笔最大亏损比例

    async def _get_market_price(self, symbol: str) -> float:
        """获取实时价格（模拟）"""
        # 实际场景从行情服务获取
        return 50000.0  # 占位

    async def execute_signal(self, signal: dict):
        """执行有效信号"""
        sym = signal['signal']['symbol']
        direction = signal['signal']['direction']
        strength = signal['signal']['strength']

        # 1. 风控检查：已有仓位不重复开仓
        if sym in self.positions:
            print(f"Already have position on {sym}, skip")
            return

        # 2. 计算仓位大小（基于信号强度和风险限制）
        price = await self._get_market_price(sym)
        position_value = self.capital * (0.1 + strength * 0.1)
        quantity = position_value / price

        # 3. 创建订单
        order = Order(
            id=f"ORD_{int(time.time())}_{sym}",
            symbol=sym,
            direction=direction,
            quantity=quantity,
            price=price,
            status='pending',
            create_time=time.time()
        )
        self.orders.append(order)

        # 4. 模拟成交（实际系统可加延迟）
        await asyncio.sleep(0.5)
        order.status = 'filled'
        order.fill_time = time.time()

        # 5. 更新持仓
        self.positions[sym] = Position(
            symbol=sym,
            direction=direction,
            entry_price=price,
            quantity=quantity,
            entry_time=time.time()
        )

        # 6. 记录交易日志
        await self._log_trade(order)

    async def update_pnl(self):
        """定期更新持仓盈亏"""
        while True:
            for sym, pos in self.positions.items():
                current_price = await self._get_market_price(sym)
                if pos.direction == 'long':
                    pos.unrealized_pnl = (current_price - pos.entry_price) * pos.quantity
                else:
                    pos.unrealized_pnl = (pos.entry_price - current_price) * pos.quantity

                # 检查止损
                if abs(pos.unrealized_pnl) > self.capital * self.risk_limit:
                    await self.close_position(sym, reason='stop_loss')

            await asyncio.sleep(5)

    async def close_position(self, symbol: str, reason: str = ''):
        """平仓"""
        pos = self.positions.pop(symbol, None)
        if pos:
            # 计算盈亏并更新资金
            self.capital += pos.unrealized_pnl
            print(f"[CLOSE] {symbol} {reason} PnL: {pos.unrealized_pnl:.2f}")

            # 发送平仓通知
            await self.redis.publish('trade_events', json.dumps({
                'type': 'close',
                'symbol': symbol,
                'pnl': pos.unrealized_pnl,
                'reason': reason
            }))

    async def run(self):
        """订阅有效信号队列"""
        await self.pubsub.subscribe('validated_signals')
        # 启动盈亏监控任务
        asyncio.create_task(self.update_pnl())

        async for message in self.pubsub.listen():
            if message['type'] != 'message':
                continue
            try:
                signal = json.loads(message['data'])
                await self.execute_signal(signal)
            except Exception as e:
                print(f"Execution error: {e}")

模块四：推送通知模块

交易信号和订单状态需要实时推送给用户。这里用 aiogram 实现 Telegram 机器人推送，同时支持邮件和本地日志。

# notifier.py
import asyncio
import logging
import json
import redis.asyncio as redis
from aiogram import Bot, Dispatcher
from aiogram.types import Message
import smtplib
from email.mime.text import MIMEText

class TradeNotifier:
    def __init__(self, telegram_token: str, chat_id: str):
        self.bot = Bot(token=telegram_token)
        self.chat_id = chat_id
        self.redis = redis.from_url("redis://localhost:6379")
        self.pubsub = self.redis.pubsub()

        # 配置日志
        logging.basicConfig(
            filename='trades.log',
            level=logging.INFO,
            format='%(asctime)s - %(message)s'
        )

    async def send_telegram(self, text: str):
        """发送Telegram消息"""
        try:
            await self.bot.send_message(chat_id=self.chat_id, text=text)
        except Exception as e:
            print(f"Telegram send error: {e}")

    async def send_email(self, subject: str, body: str):
        """发送邮件通知"""
        msg = MIMEText(body)
        msg['Subject'] = subject
        msg['From'] = 'quant@example.com'
        msg['To'] = 'user@example.com'

        with smtplib.SMTP('smtp.example.com', 587) as server:
            server.starttls()
            server.login('quant@example.com', 'password')
            server.send_message(msg)

    async def handle_trade_event(self, event: dict):
        """处理交易事件"""
        event_type = event.get('type')
        symbol = event.get('symbol', '')

        if event_type == 'open':
            msg = (f"🟢 开仓信号\n"
                   f"标的: {symbol}\n"
                   f"方向: {event['direction']}\n"
                   f"价格: {event['price']:.2f}")
            await self.send_telegram(msg)
            await self.send_email(f"[开仓] {symbol}", msg)

        elif event_type == 'close':
            msg = (f"🔴 平仓通知\n"
                   f"标的: {symbol}\n"
                   f"盈亏: {event['pnl']:.2f}\n"
                   f"原因: {event['reason']}")
            await self.send_telegram(msg)

        # 写入本地日志
        logging.info(json.dumps(event, ensure_ascii=False))

    async def run(self):
        """订阅交易事件"""
        await self.pubsub.subscribe('trade_events')
        async for message in self.pubsub.listen():
            if message['type'] != 'message':
                continue
            try:
                event = json.loads(message['data'])
                await self.handle_trade_event(event)
            except Exception as e:
                print(f"Notifier error: {e}")
