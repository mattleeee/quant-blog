异常处理精确化：避免bare except的三个坑

2026-08-18 Python自动化

写量化策略的时候，我见过太多这样的代码：

try:
    result = get_price_data(symbol)
except:
    result = None

跑回测的时候一切正常，实盘跑着跑着突然发现进程没了，或者Ctrl+C都杀不掉程序。这就是bare except埋的雷。

第一个坑：吞掉SystemExit和KeyboardInterrupt

bare except会捕获所有异常，包括SystemExit和KeyboardInterrupt。这意味着你的程序无法被正常终止。

看这个例子：

import time

def run_strategy():
    while True:
        try:
            # 模拟策略主循环
            time.sleep(1)
            print("策略运行中...")
        except:
            # 这里把KeyboardInterrupt也吞了
            print("捕获到异常，继续运行")
            continue

if __name__ == "__main__":
    run_strategy()

你按Ctrl+C，程序根本不响应。KeyboardInterrupt被except:捕获了，continue让它继续跑。在实盘环境里，这意味着你无法通过常规方式停止策略进程，只能kill -9强杀。

正确的做法：

import time

def run_strategy():
    while True:
        try:
            time.sleep(1)
            print("策略运行中...")
        except KeyboardInterrupt:
            print("\n收到停止信号，保存状态并退出")
            # 在这里保存策略状态
            break
        except Exception as e:
            print(f"运行异常: {e}")
            # 记录日志，继续运行
            continue

第二个坑：掩盖真实错误

bare except让你看不到错误是什么。量化策略出错时，你需要知道是数据获取失败、计算溢出还是API调用报错。用bare except，所有错误都变成None，你根本不知道哪里出了问题。

# 错误示范：所有错误都变成None
def fetch_price(symbol):
    try:
        data = api.get_price(symbol)
        return data
    except:
        return None

# 正确示范：精确捕获，保留错误信息
def fetch_price(symbol):
    try:
        data = api.get_price(symbol)
        return data
    except ConnectionError:
        # 网络问题，可以重试
        logger.error(f"连接失败: {symbol}")
        return None
    except TimeoutError:
        # 超时，可以调整参数
        logger.error(f"请求超时: {symbol}")
        return None
    except ValueError as e:
        # 数据格式不对，需要检查API返回
        logger.error(f"数据格式错误: {symbol}, {e}")
        raise

第三个坑：无法区分业务逻辑错误

在量化系统里，业务逻辑错误和系统错误需要分开处理。比如：

  • 系统错误：数据库连接失败、网络断开——需要重试或报警
  • 业务错误：资金不足、持仓不足——需要停止该策略或通知用户

用bare except，这两类错误混在一起，你没法针对性地处理。

class InsufficientCapital(Exception):
    """资金不足异常"""
    pass

class PositionNotFound(Exception):
    """持仓不存在异常"""
    pass

def execute_order(order):
    try:
        account = get_account()
        if account.available < order.amount:
            raise InsufficientCapital(f"可用资金不足: {account.available}")

        position = get_position(order.symbol)
        if not position and order.side == 'sell':
            raise PositionNotFound(f"没有持仓: {order.symbol}")

        # 执行下单
        return api.submit_order(order)

    except InsufficientCapital as e:
        # 业务逻辑错误，通知用户，停止策略
        logger.error(f"订单被拒绝: {e}")
        stop_strategy()
        return None
    except (ConnectionError, TimeoutError) as e:
        # 系统错误，重试
        logger.warning(f"系统异常，重试: {e}")
        return retry(order)

精确异常捕获方法论

写异常处理的核心原则：只捕获你能处理的异常，其余的让它抛出。

1. 从具体到通用

try:
    result = process_data(data)
except KeyError as e:
    # 缺少关键字段，数据格式有问题
    logger.error(f"数据缺少字段: {e}")
except (TypeError, ValueError) as e:
    # 数据类型或值不对
    logger.error(f"数据校验失败: {e}")
except Exception as e:
    # 兜底，但至少要记录完整堆栈
    logger.exception(f"未预期的错误: {e}")
    raise  # 重新抛出，让上层决定怎么处理

2. 使用else和finally

def load_config(path):
    try:
        with open(path, 'r') as f:
            config = json.load(f)
    except FileNotFoundError:
        # 配置文件不存在，用默认配置
        logger.warning(f"配置文件不存在: {path}")
        config = get_default_config()
    except json.JSONDecodeError as e:
        # JSON格式错误，需要人工干预
        logger.error(f"配置文件格式错误: {path}")
        raise ConfigError(f"配置文件解析失败: {e}")
    else:
        # 没有异常时执行
        logger.info(f"配置文件加载成功: {path}")
    finally:
        # 无论是否异常都执行
        logger.debug("配置加载流程结束")

    return config

自定义异常类设计

在量化系统中，自定义异常类让错误处理更清晰。设计原则：

  1. 继承Exception，不要继承BaseException
  2. 异常类命名要见名知义
  3. 可以在异常中携带额外的上下文信息
class StrategyError(Exception):
    """策略基础异常"""
    def __init__(self, message, strategy_id=None, **context):
        super().__init__(message)
        self.strategy_id = strategy_id
        self.context = context

class DataError(StrategyError):
    """数据相关异常"""
    pass

class OrderError(StrategyError):
    """订单执行异常"""
    def __init__(self, message, order_id=None, **kwargs):
        super().__init__(message, **kwargs)
        self.order_id = order_id

class RiskCheckError(StrategyError):
    """风控检查异常"""
    def __init__(self, message, risk_metric, limit, actual, **kwargs):
        super().__init__(message, **kwargs)
        self.risk_metric = risk_metric
        self.limit = limit
        self.actual = actual

# 使用示例
def check_risk(position, limits):
    if position.value > limits.max_value:
        raise RiskCheckError(
            f"持仓市值超限",
            risk_metric="position_value",
            limit=limits.max_value,
            actual=position.value,
            strategy_id=position.strategy_id
        )

异常日志记录方案

异常处理不只是try-except，更重要的是记录完整的错误信息。推荐使用logging模块，而不是print。

import logging
import traceback
from logging.handlers import RotatingFileHandler

# 配置日志
def setup_logger():
    logger = logging.getLogger('quant')
    logger.setLevel(logging.DEBUG)

    # 文件处理器，按大小轮转
    file_handler = RotatingFileHandler(
        'quant.log',
        maxBytes=10*1024*1024,  # 10MB
        backupCount=5,
        encoding='utf-8'
    )
    file_handler.setLevel(logging.DEBUG)

    # 控制台输出
    console_handler = logging.StreamHandler()
    console_handler.setLevel(logging.INFO)

    # 格式
    formatter = logging.Formatter(
        '%(asctime)s - %(name)s - %(levelname)s - %(filename)s:%(lineno)d - %(message)s'
    )
    file_handler.setFormatter(formatter)
    console_handler.setFormatter(formatter)

    logger.addHandler(file_handler)
    logger.addHandler(console_handler)
    return logger

logger = setup_logger()

# 在异常处理中使用
def safe_call(func, *args, **kwargs):
    try:
        return func(*args, **kwargs)
    except Exception as e:
        logger.exception(f"函数调用失败: {func.__name__}")
        # logger.exception会自动记录完整堆栈
        return None

logger.exception比logger.error更好用，它会自动带上完整的堆栈信息，排查问题的时候能省不少时间。

实战案例：一个完整的异常处理模板

下面是我在量化策略中常用的异常处理模板：

import time
from functools import wraps

def retry(max_retries=3, delay=1, exceptions=(ConnectionError, TimeoutError)):
    """重试装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    if attempt == max_retries - 1:
                        logger.error(f"重试{max_retries}次仍然失败: {e}")
                        raise
                    logger.warning(f"第{attempt + 1}次调用失败: {e}, {delay}秒后重试")
                    time.sleep(delay)
            return None
        return wrapper
    return decorator

class DataFetcher:
    def __init__(self):
        self.logger = logging.getLogger('quant.data')

    @retry(max_retries=3, delay=2)
    def fetch_kline(self, symbol, interval, limit=100):
        """获取K线数据，自动重试"""
        try:
            data = api.get_kline(symbol, interval, limit)
            if not data:
                raise DataError(f"K线数据为空: {symbol}")
            return data
        except DataError as e:
            self.logger.error(f"数据异常: {e}")
            raise
        except Exception as e:
            self.logger.exception(f"获取K线失败: {e}")
            raise

    def process_strategy(self, symbol):
        """策略主流程"""
        try:
            # 1. 获取数据
            kline_data = self.fetch_kline(symbol, '1m')

            # 2. 计算指标
            indicators = self.calculate_indicators(kline_data)

            # 3. 生成信号
            signal = self.generate_signal(indicators)

            # 4. 执行交易
            if signal:
                self.execute_trade(symbol, signal)

        except DataError as e:
            # 数据问题，跳过本轮
            self.logger.warning(f"数据异常，跳过: {e}")
        except RiskCheckError as e:
            # 风控问题，停止策略
            self.logger.critical(f"风控触发，停止策略: {e}")
            self.stop()
        except Exception as e:
            # 未知错误，记录完整堆栈
            self.logger.exception(f"策略运行未预期错误: {e}")
            # 发送告警
            self.send_alert(f"策略异常: {e}")

总结

异常处理精确化的核心就三点：

  1. 不要用bare except——它会吞掉SystemExit和KeyboardInterrupt，让程序失去控制
  2. 只捕获你能处理的异常——把异常分类处理，业务错误和系统错误分开
  3. 用logger.exception记录完整堆栈——排查问题的时候，堆栈信息比错误消息更有用

量化系统跑在无人值守的环境里，异常处理就是你最后的防线。写清楚每个异常怎么处理，比写更多的策略逻辑更重要。
