Python缓存管理：防止内存泄漏的定期清理方案

2026-08-19 Python自动化

长期运行的Python程序（比如量化交易策略、数据采集服务）最头疼的问题之一就是内存悄悄涨上去，最后OOM被杀。缓存是罪魁祸首之一——你以为在加速，实际上在漏内存。

本文直接给方案：定期清理过期条目 + 限制最大缓存数量 + LRU淘汰策略。全部有代码，直接能跑。

问题本质：缓存为什么导致内存泄漏

Python的dict不会自动清理，你往里塞数据，它就一直在内存里。量化场景最常见：

  • 缓存K线数据，越积越多
  • 缓存计算结果，key永远不重复
  • 缓存外部API响应，没设TTL

结果就是程序跑三天，内存从200MB涨到2GB。不是Python的GC不干活，是你自己把对象引用攥着不放。

方案一：基础版——手动清理过期条目

最朴素的做法：每个缓存条目记录时间戳，定期扫描删除过期数据。

import time
import threading
from typing import Any, Dict, Tuple, Optional

class TimeBasedCache:
    """带TTL的缓存，定期清理过期条目"""

    def __init__(self, ttl: int = 300, cleanup_interval: int = 60):
        """
        Args:
            ttl: 条目存活时间（秒）
            cleanup_interval: 清理间隔（秒）
        """
        self._cache: Dict[str, Tuple[Any, float]] = {}
        self._ttl = ttl
        self._cleanup_interval = cleanup_interval
        self._lock = threading.Lock()
        self._start_cleanup_thread()

    def set(self, key: str, value: Any):
        """写入缓存"""
        with self._lock:
            self._cache[key] = (value, time.time())

    def get(self, key: str) -> Optional[Any]:
        """读取缓存，过期返回None"""
        with self._lock:
            item = self._cache.get(key)
            if item is None:
                return None
            value, timestamp = item
            if time.time() - timestamp > self._ttl:
                del self._cache[key]
                return None
            return value

    def _cleanup(self):
        """清理过期条目"""
        now = time.time()
        with self._lock:
            expired = [
                key for key, (_, ts) in self._cache.items()
                if now - ts > self._ttl
            ]
            for key in expired:
                del self._cache[key]
            if expired:
                print(f"[Cache] 清理了 {len(expired)} 条过期缓存，剩余 {len(self._cache)} 条")

    def _start_cleanup_thread(self):
        """启动后台清理线程"""
        def run():
            while True:
                time.sleep(self._cleanup_interval)
                self._cleanup()

        t = threading.Thread(target=run, daemon=True)
        t.start()

    def size(self) -> int:
        return len(self._cache)

用法很简单：

# 缓存5秒，每10秒清理一次
cache = TimeBasedCache(ttl=5, cleanup_interval=10)

cache.set("btc_price", 50000)
print(cache.get("btc_price"))  # 50000

time.sleep(6)
print(cache.get("btc_price"))  # None，已过期

# 等后台清理线程跑一次
time.sleep(5)
print(f"缓存大小: {cache.size()}")  # 0

这个方案解决了“条目永远不删”的问题，但还有个隐患：如果短时间内写入大量不同key的数据，即使每个key都设了TTL，在过期之前的窗口期内，内存可能已经爆了。

方案二：进阶版——限制最大缓存数量 + LRU

光靠TTL不够，必须限制缓存容量。用collections.OrderedDict实现LRU（Least Recently Used）淘汰策略。

import time
import threading
from collections import OrderedDict
from typing import Any, Optional

class LRUCacheWithTTL:
    """LRU缓存 + TTL过期 + 定期清理"""

    def __init__(self, max_size: int = 1000, ttl: int = 300, cleanup_interval: int = 60):
        """
        Args:
            max_size: 最大缓存条目数
            ttl: 条目存活时间（秒）
            cleanup_interval: 清理间隔（秒）
        """
        self._cache: OrderedDict[str, tuple[Any, float]] = OrderedDict()
        self._max_size = max_size
        self._ttl = ttl
        self._lock = threading.Lock()
        self._cleanup_interval = cleanup_interval
        self._start_cleanup_thread()

    def get(self, key: str) -> Optional[Any]:
        """读取缓存，命中后移到末尾（表示最近使用）"""
        with self._lock:
            if key not in self._cache:
                return None
            value, timestamp = self._cache[key]
            # 检查是否过期
            if time.time() - timestamp > self._ttl:
                del self._cache[key]
                return None
            # LRU：移到末尾
            self._cache.move_to_end(key)
            return value

    def set(self, key: str, value: Any):
        """写入缓存，超容量时淘汰最久未用的"""
        with self._lock:
            if key in self._cache:
                # 更新已有key
                self._cache[key] = (value, time.time())
                self._cache.move_to_end(key)
            else:
                self._cache[key] = (value, time.time())
                # 超容量，淘汰最久未用的（头部）
                while len(self._cache) > self._max_size:
                    oldest_key, _ = self._cache.popitem(last=False)
                    print(f"[Cache] LRU淘汰: {oldest_key}")

    def _cleanup(self):
        """定期清理过期条目"""
        now = time.time()
        with self._lock:
            expired_keys = [
                key for key, (_, ts) in self._cache.items()
                if now - ts > self._ttl
            ]
            for key in expired_keys:
                del self._cache[key]
            if expired_keys:
                print(f"[Cache] 清理过期 {len(expired_keys)} 条，当前 {len(self._cache)} 条")

    def _start_cleanup_thread(self):
        def run():
            while True:
                time.sleep(self._cleanup_interval)
                self._cleanup()

        threading.Thread(target=run, daemon=True).start()

    def size(self) -> int:
        return len(self._cache)

关键点：

  • OrderedDict的move_to_end实现LRU，O(1)复杂度
  • popitem(last=False)从头部弹出最久未用的条目
  • TTL和LRU双管齐下：LRU管容量，TTL管时效

方案三：监控内存，提前报警

光清理还不够，你得知道内存到底什么情况。用psutil监控进程内存，设置阈值触发告警或额外清理。

import psutil
import os

def get_process_memory_mb() -> float:
    """获取当前进程内存占用（MB）"""
    process = psutil.Process(os.getpid())
    return process.memory_info().rss / 1024 / 1024

class MemoryMonitor:
    """内存监控器，超阈值触发回调"""

    def __init__(self, threshold_mb: float = 500, callback=None):
        """
        Args:
            threshold_mb: 内存阈值（MB）
            callback: 超阈值时的回调函数
        """
        self.threshold_mb = threshold_mb
        self.callback = callback
        self._check_interval = 5  # 每5秒检查一次
        self._start_monitor()

    def _check_memory(self):
        current = get_process_memory_mb()
        if current > self.threshold_mb:
            print(f"[Memory] 警告！内存占用 {current:.1f}MB，超过阈值 {self.threshold_mb}MB")
            if self.callback:
                self.callback(current)

    def _start_monitor(self):
        def run():
            while True:
                time.sleep(self._check_interval)
                self._check_memory()

        threading.Thread(target=run, daemon=True).start()

组合使用：

# 创建缓存：最多500条，TTL 60秒，每30秒清理
cache = LRUCacheWithTTL(max_size=500, ttl=60, cleanup_interval=30)

# 内存监控：超过300MB时清空缓存
def emergency_cleanup(memory_mb):
    print(f"[Emergency] 内存 {memory_mb:.1f}MB，清空缓存！")
    cache._cache.clear()  # 直接清空底层dict

monitor = MemoryMonitor(threshold_mb=300, callback=emergency_cleanup)

# 模拟写入数据
for i in range(10000):
    cache.set(f"key_{i}", {"data": "x" * 1024})  # 每条1KB
    if i % 100 == 0:
        time.sleep(0.01)

print(f"最终缓存大小: {cache.size()}")

实战建议：量化场景怎么用

  1. 行情数据缓存：K线数据设TTL 60秒，最多缓存最近1000根K线，防止历史数据堆积。

  2. 指标计算结果：技术指标（MA、RSI等）的计算结果可以缓存，但必须设置TTL，因为新K线来了旧值就失效。

  3. API响应缓存：外部数据源（如交易所REST API）的响应，TTL设5-10秒，避免频繁请求，同时保证数据新鲜度。

  4. 回测优化：回测中缓存因子值，用LRU限制大小，防止因子矩阵爆炸。

完整示例：整合到你的策略里

class QuantStrategy:
    """带缓存管理的策略基类"""

    def __init__(self, symbol: str):
        self.symbol = symbol
        # K线缓存：最多1000条，TTL 300秒
        self.kline_cache = LRUCacheWithTTL(
            max_size=1000, 
            ttl=300, 
            cleanup_interval=60
        )
        # 指标缓存：最多200条，TTL 60秒
        self.indicator_cache = LRUCacheWithTTL(
            max_size=200,
            ttl=60,
            cleanup_interval=30
        )
        # 内存监控：超过500MB报警
        self.memory_monitor = MemoryMonitor(
            threshold_mb=500,
            callback=self._emergency_cleanup
        )

    def _emergency_cleanup(self, memory_mb):
        """紧急清理：先清指标缓存，再清K线缓存"""
        print(f"[Emergency] 清空指标缓存")
        self.indicator_cache._cache.clear()
        if memory_mb > 700:  # 更严重才清K线
            print(f"[Emergency] 清空K线缓存")
            self.kline_cache._cache.clear()

    def get_kline(self, symbol: str, timeframe: str = "1m"):
        """获取K线，带缓存"""
        cache_key = f"{symbol}_{timeframe}"
        data = self.kline_cache.get(cache_key)
        if data is None:
            # 这里写你的数据获取逻辑
            data = self._fetch_kline(symbol, timeframe)
            self.kline_cache.set(cache_key, data)
        return data

    def _fetch_kline(self, symbol: str, timeframe: str):
        """模拟获取K线数据"""
        return {"symbol": symbol, "timeframe": timeframe, "data": []}

总结

缓存管理不是可选项，是长期运行程序的刚需。核心就三点：

  1. TTL保证时效性——过期数据不占内存
  2. LRU保证容量——最多存多少条，超出就淘汰
  3. 内存监控兜底——异常情况能紧急清理
